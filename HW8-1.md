### 6.71
假设树上每个结点所含的数据元素为一个字母，并且以孩子-兄弟链表为树的存储结构，试写一个按凸入表方式打印一棵树的算法。例如：左下所示树印为右下形状

原二叉树：

```text
        A
      /  | \
    B    C   D
  /   \  |   
  E    F G 
           
            
```

树状打印结果（右子树在上、左子树在下）：

```text
    A
        B   
            E
            F
        C
            G
        D
```

```c
void PrintTree(CSTree T, int depth){
    if(T==NULL){
        return;
    }
    for(int i=0;i<depth;i++){
        printf("    ");
    }
    printf("%c/n",T->data);
    for(CSTree p=T->firstchild;p!=NULL;pp->nextsibling){
        PrintTree(p,depth+1);
    }
}
```

### 7.15
试在邻接矩阵存储结构上实现图的基本操作：`InsertVex(G,v)` , `InsertArc(G,v,w)`  ,  `DeleteVex(G,v)`  ,  `DeleteArc(G,v,w)` 

```c
typedef struct{
    ElemType vexs[MAX_VERTEX_NUM];
    int arcs[MAX_VERTEX_NUM][MAX_VERTEX_NUM];
    int vecnum;
    int arcnum;
    int kind; //区分有向或者无向图
} MGraph;

int LocateVex(MGraph *G,ElemType v){
    for(int i=0;i<G->vecnum;i++){
        if(G->vexs[i]==v){
            return i;
        }
    }
    return -1;
}

Status InsertVex(MGraph *G,ElemType v){
    if(G->vexnum>=MAX_VERTEX_NUM){
        return ERROR;
    }
    if(LocateVex(G,v)!=-1){
        return ERROR;
    }
    int n=G->vexnum;
    G->vexs[n]=v;
    for(int i=0;i<=n;i++){
        G->arcs[n][i]=0;
        G->arcs[i][n]=0;
    }
    G->vecnum++;
    return OK;
}

Status DeleteVex(MGraph *G,ElemType v){
    int k=LocateVex(G,v);
    if(k==-1){
        return -1;
    }
    int n=G->vecnum;
    for(int i=k;i<n-1;i++){
        G->vexs[i]=G->vexs[i+1]
        for(int j=0;j<n;j++){
            G->arcs[i][j]=G->[i+1][j];
        }
    }
    G->vecnum-=1
    for(int i=0;i<G->vecnum;i++){
        G->arcs[G->vecnum][i]=0;
        G->arcs[i][G->vecnum]=0;
    }
    G->srcs[G->vecnum][G->vecnum]=0
    return 1;
}

Status InsertArc(MGraph *G,ElemType v,ElemType w){
    int i=LocateVex(G,v);
    int j=LocateVex(G,w);
    if(i==-1||j==-1){
        return -1;
    }
    if(G->arcs[i][j]){
        return -1;
    }
    G->arcs[i][j]=1;
    G->arcnum+=1;
    return 1;
}

Status DeleteArc(MGraph *G,ElemType v,ElemType w){
    int i=LocateVex(G,v);
    int j=LocateVex(G,w);
    if(i==-1||j==-1){
        return -1;
    }
    if(!G->arcs[i][j]){
        return -1;
    }
    G->arcs[i][j]=0;
    G->arcnum--;
    return 1;
}
```

### 7.16
试对邻接表存储结构重做 **7.15** 题
```c
typedef struct ArcNode {
    int adjvex;                
    struct ArcNode *nextarc;   
} ArcNode;

typedef struct VNode {
    VertexType data;           
    ArcNode *firstarc;          
} VNode, AdjList[MAX_VERTEX_NUM];

typedef struct {
    AdjList vertices;         
    int vexnum, arcnum;         
    int kind;                   
} ALGraph;


int LocateVex(ALGraph *G, VertexType v) {
    for (int i = 0; i < G->vexnum; i++)
        if (G->vertices[i].data == v)
            return i;
    return -1;
}

Status InsertVex(ALGraph *G, VertexType v) {
    if (G->vexnum >= MAX_VERTEX_NUM)
        return ERROR;
    if (LocateVex(G, v) != -1)         
        return ERROR;

    G->vertices[G->vexnum].data = v;
    G->vertices[G->vexnum].firstarc = NULL;
    G->vexnum++;
    return OK;
}


Status InsertArc(ALGraph *G, VertexType v, VertexType w) {
    int i = LocateVex(G, v);
    int j = LocateVex(G, w);
    if (i == -1 || j == -1)
        return ERROR;
    for (ArcNode *cur = G->vertices[i].firstarc; cur != NULL; cur = cur->nextarc)
        if (cur->adjvex == j)
            return ERROR;

    ArcNode *p = (ArcNode *)malloc(sizeof(ArcNode));
    ArcNode *q = NULL;
    p->adjvex = j;
    p->nextarc = G->vertices[i].firstarc;
    G->vertices[i].firstarc = p;
    G->arcnum++;
    return OK;
}

static int RemoveArc(ALGraph *G, int i, int j) {
    ArcNode **pp = &G->vertices[i].firstarc;

    while (*pp != NULL) {
        if ((*pp)->adjvex == j) {
            ArcNode *q = *pp;
            *pp = q->nextarc;
            free(q);
            return 1;
        }
        pp = &(*pp)->nextarc;
    }
    return 0;
}
Status DeleteArc(ALGraph *G, VertexType v, VertexType w) {
    int i = LocateVex(G, v);
    int j = LocateVex(G, w);
    if (i == -1 || j == -1)
        return ERROR;

    if (!RemoveArc(G, i, j))
        return ERROR;
    G->arcnum--;
    return OK;
}


Status DeleteVex(ALGraph *G, VertexType v) {
    int k = LocateVex(G, v);
    if (k == -1)
        return ERROR;
    for (int i = 0; i < G->vexnum; i++) {
        if (i == k)
            continue;
        ArcNode **pp = &G->vertices[i].firstarc;
        while (*pp != NULL) {
            ArcNode *cur = *pp;
            if (cur->adjvex == k) {
                *pp = cur->nextarc;
                free(cur);
            } else {
                if (cur->adjvex > k)
                    cur->adjvex--;
                pp = &cur->nextarc;
            }
        }
    }
    ArcNode *cur = G->vertices[k].firstarc;
    while (cur != NULL) {
        ArcNode *next = cur->nextarc;
        free(cur);
        cur = next;
    }
    for (int i = k; i < G->vexnum - 1; i++)
        G->vertices[i] = G->vertices[i + 1];
    G->vexnum--;
    G->arcnum = 0;
    for (int i = 0; i < G->vexnum; i++)
        for (ArcNode *p = G->vertices[i].firstarc; p != NULL; p = p->nextarc)
            G->arcnum++;
    return OK;
}
```
