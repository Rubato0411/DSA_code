假设以块链结构表示串，试编写将串`s`插入到串`t`中某个字符之后的算法（若串`t`中不存在次字符，则将串`s`连接在串`t`的末尾）。

```c
void Insert(LString* t,LString* s,char c){
    Chunk* p=t->head;
    int flag=0;
    int i=0;
    int j=0;
    int k=0;
    if(s==NULL||s->head==NULL){
        return;
    }
    if(t->head==NULL) {

            }
        }

            p=p->next;
        }
    }
    k=i+1;
    int m=i+1;
    if(flag==0){
        t->tail->next=s->head;
        t->tail=s->tail;
    }else{
        Chunk* cur=p;
        if(i==CHUNKSIZE-1){
            s->tail->next=p->next;
            p->next=s->head;
            if(p==t->tail){
                t->tail=s->tail;
            }
        }else{
            for(k=i+1;k<CHUNKSIZE;k++){
                extra->ch[j]=p->ch[k];
                j++;
            }
            while(j<CHUNKSIZE){
                extra->ch[j]='#';
                j++;
            }
            while(m<CHUNKSIZE){
                p->ch[m]='#';
                m++;

            extra->next=p->next;
            p->next=extra;
            s->tail->next=extra;
            p->next=s->head;
        }
    }
    t->curlen+=s->curlen;
}
```

```c
void commonString(LString* s,LString* t,LString* result){
    int dp[MAXSTRLEN+1][MAXSTRLEN+1]={0};
    int i=0;
    int j=0;
    int maxlen=0;
    int index=0;
    int k=0;
    for(i=1;i<=s->length;i++){
        for(j=1;j<=t->length;j++){
            if(s->ch[i]==t->ch[j]){
                dp[i][j]=dp[i-1][j-1]+1;
                if(dp[i][j]>maxlen){
                maxlen=dp[i][j];
                index=j;
                }
            }else{
                dp[i][j]=0;
            }
        }
    }
    if(maxlen==0){
        result->length=0;
        return;
    }
    for(j=index-maxlen+1,k=1;j<=index;j++,k++){
        result->ch[k]=t->ch[j];
    }
    result->length=maxlen;
}
```
